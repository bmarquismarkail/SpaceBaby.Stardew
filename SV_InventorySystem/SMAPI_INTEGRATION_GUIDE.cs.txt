using StardewModdingAPI;
using StardewValley;
using SV_InventorySystem.Framework.Reflection;

namespace YourMod;

/// <summary>Minimal integration with the public SV_InventorySystem API.</summary>
public sealed class InventoryIntegration
{
    private const string InventoryModId = "SpaceBaby.SV_InventorySystem";

    private readonly IModHelper helper;
    private readonly IMonitor monitor;
    private IMultiInventoryManager? inventoryManager;

    public InventoryIntegration(IModHelper helper, IMonitor monitor)
    {
        this.helper = helper;
        this.monitor = monitor;
    }

    /// <summary>Call after all mods have been loaded, such as from GameLaunched.</summary>
    public void Initialize()
    {
        this.inventoryManager = this.helper.ModRegistry.GetApi<IMultiInventoryManager>(InventoryModId);
        if (this.inventoryManager == null)
        {
            this.monitor.Log(
                "SV_InventorySystem isn't installed or its required patches aren't active. Multi-inventory features are disabled.",
                LogLevel.Warn);
        }
    }

    public Item? GetCurrentItem(Farmer farmer)
    {
        return this.inventoryManager?.GetCurrentItem(farmer) ?? farmer.CurrentItem;
    }

    public bool TrySwitchInventory(Farmer farmer, int inventoryIndex)
    {
        if (this.inventoryManager?.GetInventory(farmer, inventoryIndex) == null)
            return false;

        this.inventoryManager.SetActiveInventoryIndex(farmer, inventoryIndex);
        return this.inventoryManager.GetActiveInventoryIndex(farmer) == inventoryIndex;
    }

    public Item? AddAtSlot(Farmer farmer, Item item, int inventoryIndex, int localIndex)
    {
        int? globalIndex = this.inventoryManager?.GetGlobalIndex(farmer, inventoryIndex, localIndex);
        return globalIndex == null
            ? item
            : this.inventoryManager!.AddItemAtIndexWithRemainder(farmer, item, globalIndex.Value);
    }
}
